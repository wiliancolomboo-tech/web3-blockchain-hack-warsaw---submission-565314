FROM node:20-bullseye-slim

WORKDIR /app

# Install system dependencies
RUN apt-get update && apt-get install -y --no-install-recommends \
    curl build-essential git \
    && rm -rf /var/lib/apt/lists/*

COPY client/package*.json ./
RUN npm ci

COPY client/ ./
RUN npm run build

EXPOSE 8080
CMD ["npm", "run", "start"]
